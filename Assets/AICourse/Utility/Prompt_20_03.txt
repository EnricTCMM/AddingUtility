// --- DYNAMIC BLACKBOARD (Fragment rellevant per a Utility AI) ---
public class DynamicBlackboard : MonoBehaviour {
    // Guarda els límits llegits de l'atribut [Range(min, max)] de Unity via Reflexió
    private Dictionary<string, (float min, float max)> propertyRanges = new Dictionary<string, (float, float)>();
    
    public T Get<T>(string key) { /* ... */ }
    
    public bool TryGetRange(string key, out float min, out float max) {
        if (!initialised) Initialize();
        key = key.ToUpper();
        if (propertyRanges.TryGetValue(key, out var range)) {
            min = range.min; max = range.max; return true;
        }
        min = 0; max = 100; return false; // Fallback intel·ligent per defecte
    }
}

// --- INTERFÍCIE BASE ---
namespace Utility {
    public interface IConsideration {
        string Name { get; } 
        void Contextualize(GameObject go);
        float GetScore();
    }

// --- CLASSE CONSIDERATION (Avaluador de propietats - Tancat) ---
    public class Consideration : IConsideration {
        public string Name { get; set; }
        public GameObject gameObject;
        public DynamicBlackboard blackboard;
        
        private Func<float, float> responseCurve;
        private Func<float> syntheticProperty;
        private string propertyKey; 
        private float propertyMin, propertyMax;
        private bool explicitRangeProvided = false;

        // Constructor 1: Propietats de la Pissarra
        public Consideration(string key, Func<float, float> responseCurve, string name = null) {
            this.propertyKey = key; this.responseCurve = responseCurve;
            this.explicitRangeProvided = false;
            Name = name == null ? "Consideration("+key+")" : name + "("+key+")";
        }

        // Constructor 2: Propietats Sintètiques (Lambda)
        public Consideration(Func<float> syntheticProperty, Func<float, float> responseCurve, float min, float max, string name = null) {
            this.syntheticProperty = syntheticProperty; this.responseCurve = responseCurve;
            this.explicitRangeProvided = true; propertyMin = min; propertyMax = max;
            Name = name == null ? "Synth. Consideration" : name;
        }
        
        public void Contextualize(GameObject go) {
            gameObject = go; blackboard = gameObject.GetComponent<DynamicBlackboard>();
            if (!explicitRangeProvided) {
                bool hasRange = blackboard.TryGetRange(propertyKey, out propertyMin, out propertyMax);
                if(!hasRange) Debug.LogWarning("Consideration " + propertyKey + " is not ranged. Using [0, 100]."); 
            }
            if (propertyMin == propertyMax) Debug.LogError("Consideration " + propertyKey + " has zero range.");
        }

        public float GetScore() {
            float val = syntheticProperty != null ? syntheticProperty() : blackboard.Get<float>(propertyKey);
            val = (val - propertyMin) / (propertyMax - propertyMin); // Normalització [0,1]
            return Mathf.Clamp01(responseCurve(Mathf.Clamp01(val)));
        }
    }
}
