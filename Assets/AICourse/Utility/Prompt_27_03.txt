Rol: Actua com un expert en Arquitectura de Software per a Videojocs, Unity (C#) i Intel·ligència Artificial, amb un enfocament altament docent i pedagògic.

Context del Projecte:
Estic desenvolupant un framework de Utility AI (IA d'Utilitat) en Unity per als meus alumnes. La filosofia de disseny és "Code as Configuration": vull que els alumnes dissenyin els arbres d'IA purament per codi C# mitjançant instanciació neta, evitant configuracions feixugues a l'Inspector o diccionaris complexos. Les APIs han de ser semàntiques, molt llegibles, segures (amb proteccions fail-fast que aturin el joc si hi ha errors de disseny) i optimitzades pel que fa al Garbage Collection (evitar allocacions innecessàries per frame). Aquest framework s'integrarà amb un sistema d'Arbres de Comportament (Behavior Trees) que ja tinc construït (namespace BTs).

La Tasca Immediata:
Ara necessito dissenyar el "Cervell" de l'Agent. Ha de ser un component MonoBehaviour (tipus UtilityBrain o AIAgent) que s'enganxi a l'NPC en Unity. Aquest component ha de rebre l'UtilityActionSet, inicialitzar-lo passant el propi GameObject via Contextualize(), i tenir el "Main Loop" (a l'Update o similar) on s'iteri per les accions, s'avaluïn les puntuacions dels Scorers, es triï l'acció guanyadora i s'enviï al framework del Behavior Tree per executar-la.

Si us plau, confirma'm que has entès l'arquitectura i la filosofia docent, i proposa'm la primera versió d'aquest MonoBehaviour (El Cervell).

using System;
using UnityEngine;

namespace Utility
{
    public static class Curves
    {
        public static Func<float, float> Exponential(float intensity = 2f) 
        { return (x) => Mathf.Pow(x, intensity); }

        public static Func<float, float> Logarithmic(float intensity = 2f) 
        { 
            float safeIntensity = Mathf.Max(0.0001f, intensity);
            return Exponential(1f / safeIntensity); 
        }

        public static Func<float, float> Sigmoid(float steepness = 10f, float midpoint = 0.5f)
        { return (x) => Mathf.Clamp01(1f / (1f + Mathf.Exp(-steepness * (x - midpoint)))); }

        public static Func<float, float> InvertedExponential(float intensity = 2f)
        {
            Func<float, float> baseCurve = Exponential(intensity);
            return (x) => 1f - baseCurve(x);
        }

        public static Func<float, float> InvertedLogarithmic(float intensity = 2f)
        {
            float safeIntensity = Mathf.Max(0.0001f, intensity);
            return InvertedExponential(1f / safeIntensity);
        }

        public static Func<float, float> InvertedSigmoid(float steepness = 10f, float midpoint = 0.5f)
        {
            Func<float, float> baseCurve = Sigmoid(steepness, midpoint);
            return (x) => 1f - baseCurve(x);
        }

        public static readonly Func<float, float> MILD_EXPONENTIAL = Exponential(1.5f); 
        public static readonly Func<float, float> MEDIUM_EXPONENTIAL = Exponential(2f);

2. Scorer.cs (Avaluador d'Accions)
C#
using System.Collections.Generic;
using UnityEngine;

namespace Utility
{
    public class Scorer : IConsideration
    {
        public string Name { get; set; }
        public GameObject gameObject;
        public DynamicBlackboard blackboard;
        
        private List<IConsideration> considerations = new List<IConsideration>();

        public AggregationPolicy policy = AggregationPolicy.ADJUSTED_MULTIPLY;
        
        public void Contextualize(GameObject go)
        {
            gameObject = go;
            blackboard = gameObject.GetComponent<DynamicBlackboard>();
            foreach (IConsideration consideration in considerations)
            { 
                consideration.Contextualize(gameObject);
            }
        }

        public Scorer(string name, AggregationPolicy policy, params IConsideration[] considerations)
        {
            Name = name;
            this.policy = policy;
            AddConsiderations(considerations);
        }
        
        public void AddConsideration(IConsideration c)
        {
            considerations.Add(c);
        }

        public void AddConsiderations(params IConsideration[] considerations)
        {
            foreach (IConsideration consideration in considerations)
            {
                AddConsideration(consideration);
            }
        }

        public float GetScore()
        {
            if (considerations.Count == 0) 
            {
                throw new System.InvalidOperationException("Cannot get score from scorer " + Name + " because it has no considerations");
            }
            
            switch (policy)
            {
                case AggregationPolicy.MULTIPLY:
                    return AggregateMultiply();
                case AggregationPolicy.ADJUSTED_MULTIPLY:
                    return AggregateAdjustedMultiply();
                case AggregationPolicy.AVERAGE:
                    return AggregateAverage();
                default:
                    return 0;
            }
        }

        private float AggregateMultiply()
        {
            float product = 1;
            foreach (IConsideration consideration in considerations)
            {
                product *= consideration.GetScore();
                if (product == 0) return 0f;
            }
            return product;
        }

        private float AggregateAdjustedMultiply()
        {
            float product = AggregateMultiply();
            float modFactor = 1f - (1f / considerations.Count);
            float makeupValue = (1f - product) * modFactor;
            return product + (makeupValue * product);
        }

        private float AggregateAverage()
        {
            float sum = 0;
            foreach (IConsideration consideration in considerations)
            {
                sum += consideration.GetScore();
            }
            return sum / considerations.Count;
        }
    }
}
3. UtilityActionSet.cs (Definició del comportament)
C#
using System.Collections.Generic;
using UnityEngine;
using BTs; // Framework extern de Behavior Trees

namespace Utility
{
    public abstract class UtilityActionSet : ScriptableObject
    {
        public List<ActionConsiderationPair> list = new List<ActionConsiderationPair>();

        public void Contextualize(GameObject go)
        {
            foreach (ActionConsiderationPair pair in list)
            {
                pair.consideration.Contextualize(go);
            }
        }

        private ActionConsiderationPair GetPair(Action action)
        {
            return list.Find(pair => pair.action == action);
        }
        
        private void Bind(Action action, IConsideration consideration)
        {
            if (action == null) 
                throw new System.ArgumentNullException("action", "Action cannot be null in Bind");
            if (consideration == null) 
                throw new System.ArgumentNullException("consideration", "Consideration cannot be null in Bind");
            
            if (GetPair(action).action == null)
            {
                list.Add(new ActionConsiderationPair(action, consideration));
            }
            else throw new System.ArgumentException("Action " + action.Name + " already bound to a consideration");
        }
        
        public virtual void OnConstruction()
        {
        }
    }
    
    public struct ActionConsiderationPair 
    {
        public Action action;
        public IConsideration consideration; 
        
        public ActionConsiderationPair(Action action, IConsideration consideration)
        {
            this.action = action;
            this.consideration = consideration;
        }
    }
}
